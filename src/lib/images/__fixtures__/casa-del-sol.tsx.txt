/* Each section renders inside its own boundary: one that crashes is left out and reported; the rest of the page still shows. */
class SectionBoundary extends React.Component<{ name: string; children?: React.ReactNode }, { failed: boolean }> {
  state = { failed: false };
  static getDerivedStateFromError() {
    return { failed: true };
  }
  componentDidCatch(error: any, info: any) {
    try {
      window.parent.postMessage({ type: 'IFRAME_ERROR', kind: 'runtime', partial: true, section: this.props.name, message: String((error && error.message) || error), stack: String((info && info.componentStack) || '') }, '*');
    } catch (e) {}
  }
  render() {
    // display: contents — the marker names the section for the layout audit without taking part in layout.
    return this.state.failed ? <div data-failed-section={this.props.name} /> : <div data-section={this.props.name} style={{ display: 'contents' }}>{this.props.children}</div>;
  }
}

/* IntentScene: the user's drawing, rendered by IntentDraw (papercut) — geometry, layering, colour and motion are fixed. */
const IntentScene = ({ className = '' }: { className?: string }) => {
  const root = React.useRef<HTMLDivElement | null>(null);
  React.useEffect(() => {
    const host = root.current;
    const w = window as any;
    if (!host || w.__intentdrawFrozen) return;
    if (window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches) return;
    const rise = Array.from(host.querySelectorAll('[data-rise]')) as any[];
    const draws = (Array.from(host.querySelectorAll('[data-draw]')) as any[])
      .map(p => { let len = 0; try { len = Math.ceil(p.getTotalLength()); } catch (e) {} return { p, len }; })
      .filter(d => d.len > 0);
    const layers = Array.from(host.querySelectorAll('[data-depth]')) as any[];
    const fine = !!(window.matchMedia && window.matchMedia('(pointer: fine)').matches);
    const ease = (t: number) => 1 - Math.pow(1 - Math.min(1, Math.max(0, t)), 3);
    let start = 0, intro = true, mx = 0, my = 0, x = 0, y = 0, s = 0, raf = 0;
    const settle = () => {
      intro = false;
      rise.forEach(g => { g.style.transform = ''; g.style.opacity = ''; });
      draws.forEach(d => { d.p.style.strokeDasharray = ''; d.p.style.strokeDashoffset = ''; });
    };
    const tick = (now: number) => {
      raf = 0;
      if (w.__intentdrawFrozen) { settle(); return; }
      if (intro) {
        if (!start) start = now;
        const t = now - start;
        let done = true;
        rise.forEach((g, i) => {
          const k = ease((t - 140 * i) / 1600);
          if (k < 1) done = false;
          g.style.transform = 'translateY(' + (36 * (1 - k)).toFixed(2) + 'px)';
          g.style.opacity = String(0.2 + 0.8 * k);
        });
        draws.forEach((d, i) => {
          const k = ease((t - 250 - 110 * i) / 2200);
          if (k < 1) done = false;
          d.p.style.strokeDasharray = d.len + ' ' + d.len;
          d.p.style.strokeDashoffset = String(d.len * (1 - k));
        });
        if (done) settle();
      }
      const target = Math.min(3000, window.scrollY || 0);
      x += (mx - x) * 0.07; y += (my - y) * 0.07; s += (target - s) * 0.12;
      for (const g of layers) {
        const d = Number(g.getAttribute('data-depth')) || 0;
        const lift = s * 0.3 * (1 - d);
        g.style.transform = 'translate(' + (-x * 28 * d).toFixed(2) + 'px,' + (lift - y * 10 * d).toFixed(2) + 'px) scale(' + (1 + 0.04 * d).toFixed(3) + ')';
      }
      if (intro || Math.abs(mx - x) + Math.abs(my - y) > 0.002 || Math.abs(target - s) > 0.5) raf = requestAnimationFrame(tick);
    };
    const kick = () => { if (!raf) raf = requestAnimationFrame(tick); };
    // A hidden frame can report a zero-sized window: never divide by it.
    const onMove = (e: MouseEvent) => { mx = e.clientX / Math.max(1, window.innerWidth) - 0.5; my = e.clientY / Math.max(1, window.innerHeight) - 0.5; kick(); };
    window.addEventListener('scroll', kick, { passive: true });
    if (fine) window.addEventListener('mousemove', onMove, { passive: true });
    kick();
    return () => {
      cancelAnimationFrame(raf);
      window.removeEventListener('scroll', kick);
      window.removeEventListener('mousemove', onMove);
    };
  }, []);
  return (
    <div ref={root} aria-hidden="true" data-tones="" className={`intent-scene relative block w-full h-full overflow-hidden ${className}`}>
      <svg
        viewBox="0 0 1280 800"
        preserveAspectRatio="xMidYMid slice"
        className="absolute inset-0 block w-full h-full"
      >
        <defs>
          <linearGradient id="is-sky" x1="0" y1="0" x2="0" y2="1"><stop offset="0%" stopColor="#FBD6AF" /><stop offset="100%" stopColor="#FAF3E0" /></linearGradient>
          <filter id="is-paper" x="-10%" y="-20%" width="120%" height="140%"><feDropShadow dx="0" dy="-4" stdDeviation="6" floodColor="#000000" floodOpacity="0.26" /></filter>
        </defs>
        <rect y="0" width="1280" height="800" fill="url(#is-sky)" />
        <g data-depth="1" style={{ transformOrigin: '640px 800px' }}>
          <g data-rise="1">
            {/* TacoScrollContainer */}
            <path id="is0-track" d="M13 423 L1279 423 L1279 680 L13 680" fill="none" stroke="none" />
            <circle r="7" fill="#FF6F00"><animateMotion dur="8s" repeatCount="indefinite" rotate="auto"><mpath href="#is0-track" /></animateMotion></circle>
          </g>
        </g>
      </svg>
      <div className="pointer-events-none absolute inset-0" style={{ opacity: 0.1, mixBlendMode: 'soft-light', backgroundImage: "url('data:image/svg+xml;utf8,%3Csvg%20xmlns%3D%22http%3A%2F%2Fwww.w3.org%2F2000%2Fsvg%22%20width%3D%22180%22%20height%3D%22180%22%3E%3Cfilter%20id%3D%22g%22%3E%3CfeTurbulence%20type%3D%22fractalNoise%22%20baseFrequency%3D%22.8%22%20numOctaves%3D%223%22%20stitchTiles%3D%22stitch%22%2F%3E%3CfeColorMatrix%20values%3D%220%200%200%200%20.5%200%200%200%200%20.5%200%200%200%200%20.5%200%200%200%201%200%22%2F%3E%3C%2Ffilter%3E%3Crect%20width%3D%22100%25%22%20height%3D%22100%25%22%20filter%3D%22url(%23g)%22%2F%3E%3C%2Fsvg%3E')" }} />
    </div>
  );
};
